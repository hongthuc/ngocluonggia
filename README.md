        // 1. Tự động trượt 2 Banner chuyên nghiệp
        let currentBannerIndex = 0;
        const totalBanners = 2;
        let bannerInterval = null;

        function updateBannerDisplay() {
            const slides = document.querySelectorAll('.hero-slide');
            slides.forEach((slide, idx) => {
                if (idx === currentBannerIndex) {
                    slide.classList.add('active');
                } else {
                    slide.classList.remove('active');
                }
            });

            const dot0 = document.getElementById('banner-dot-0');
            const dot1 = document.getElementById('banner-dot-1');
            if (dot0 && dot1) {

                    dot1.className = 'w-2.5 h-2 rounded-full bg-white/60 transition-all';
                } else {
                    dot0.className = 'w-2.5 h-2 rounded-full bg-white/60 transition-all';
                    dot1.className = 'w-6 h-2 rounded-full bg-amber-400 transition-all';

            }

            currentBannerIndex = (currentBannerIndex + 1) % totalBanners;
            updateBannerDisplay();
            resetBannerTimer();

            currentBannerIndex = (currentBannerIndex - 1 + totalBanners) % totalBanners;
            updateBannerDisplay();
            resetBannerTimer();

            currentBannerIndex = index;
